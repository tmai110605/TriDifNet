\newcommand{\lineagecard}[5]{%
  % #1 = tâm x, #2 = màu, #3 = tiêu đề, #4 = nội dung, #5 = tên node
  \node[anchor=north, fill=#2, text=white, rounded corners=6pt, minimum width=9.6cm,
        minimum height=1.2cm, font=\Large\bfseries, inner sep=4pt] (h#5) at (#1,-38.6) {#3};
  \node[anchor=north, draw=#2, fill=#2!8, line width=1.6pt, rounded corners=6pt,
        text width=9.0cm, align=left, inner sep=8pt, minimum height=8.6cm, font=\large]
        at ($(h#5.south)+(0,-0.2)$) {#4};
}
\lineagecard{4.9}{colTick}{Kế thừa từ TickNet (T)}{%
\begin{itemize}[leftmargin=1.1em,nosep,itemsep=3pt]
 \item \textbf{Khung xương}: Data-BN $\to$ Stem $\to$ 5 stage $\to$ Final Conv $\to$ GAP $\to$ FC
 \item \textbf{Khối FR-PDP}: Pw1 (không BN/act) $\to$ Dw$3{\times}3$ $\to$ Pw2
 \item \textbf{Full-Residual}: nhánh PwR khi $s{=}2$ hoặc đổi số kênh, ngược lại identity
 \item \textbf{SE} (cổng kênh toàn cục) $\to$ tái sử dụng làm nhánh kênh của TASA
 \item Khởi tạo Kaiming cho Conv
\end{itemize}}{tick}
\lineagecard{15.1}{colNet}{Kế thừa từ NetTOP (N)}{%
\begin{itemize}[leftmargin=1.1em,nosep,itemsep=3pt]
 \item \textbf{Ý tưởng đa mặt phẳng} $(xy,\,xz,\,yz)$: xử lý đặc trưng theo nhiều trục thay vì chỉ không gian
 \item \textbf{Cổng nhân} dạng $\mathrm{ReLU}(a\odot\sigma(b\odot c))$
 \item \textbf{Conv$1{\times}1$ nhóm} (\texttt{groups}) giảm tham số $\to$ dùng cho Pw1 khi $C\ge192$
 \item Mỗi block: Residual $+$ SE
\end{itemize}}{net}
\lineagecard{25.3}{colAtt}{Kế thừa từ FullAttNet (F)}{%
\begin{itemize}[leftmargin=1.1em,nosep,itemsep=3pt]
 \item \textbf{Multi-stride Dw} + nội suy \textit{nearest} (Dw1/Dw2 $\to$ Upsample)
 \item \textbf{Cổng nhân đa tỉ lệ} $\mathrm{ReLU}(x_1\odot\sigma(x_2\odot x_3))$
 \item \textbf{Thống kê bậc 1--2} (avg/std) và attention theo 3 trục
 \item \textbf{Dropout2d} $p{=}0.2$ trước/sau GAP; \texttt{width\_multiplier}
\end{itemize}}{att}
\lineagecard{35.5}{colNew}{Đóng góp MỚI của TriDiffNet}{%
\begin{itemize}[leftmargin=1.1em,nosep,itemsep=3pt]
 \item \textbf{TASA}: strip pooling mean$\oplus$std theo $H$/$W$ + Conv 1D ($3{\times}1$, $1{\times}3$) + cộng broadcast $\to$ cổng không gian
 \item \textbf{TensorDiff3D}: sai phân rời rạc $\Delta_H,\Delta_W,\Delta_C$ + $\tanh$ + $\alpha$ học được (khởi tạo 0)
 \item \textbf{Độc lập độ phân giải}: bỏ các bank MLP theo kích thước (112/56/28/\ldots) của FullAttNet
 \item \textbf{Rút gọn} 3 $\to$ 2 nhánh Dw: $\mathrm{ReLU}(x_1\odot\sigma(x_2))$
 \item Cấu hình \textit{compact}/\textit{ultra} $\approx$1,5M / 1,05M tham số
\end{itemize}}{newc}
